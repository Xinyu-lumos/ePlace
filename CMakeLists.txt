cmake_minimum_required(VERSION 3.16)
project(ePlace LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(Threads REQUIRED)
find_package(OpenMP QUIET)
add_library(eplace_platform INTERFACE)
target_link_libraries(eplace_platform INTERFACE Threads::Threads ${CMAKE_DL_LIBS})
if(OpenMP_CXX_FOUND)
    target_link_libraries(eplace_platform INTERFACE OpenMP::OpenMP_CXX)
endif()
# Plotting writes BMP files and does not need an X11 display.
target_compile_definitions(eplace_platform INTERFACE cimg_display=0)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(eplace_platform INTERFACE -fpermissive)
endif()

add_subdirectory(QPlace)
add_subdirectory(Parser)
add_subdirectory(EPlace)
add_subdirectory(Optimization)
add_subdirectory(Legalization)
add_subdirectory(main)
